```cpp
#include <iostream>
using namespace std;

struct Node {
    int key;
    Node* left;
    Node* right;
};

Node* insert(Node* root, int key) {
    if (root == nullptr) {
        return new Node{key, nullptr, nullptr};
    }
    if (key < root->key) {
        root->left = insert(root->left, key);
    } else {
        root->right = insert(root->right, key);
    }
    return root;
}

void inorder(const Node* node) {
    if (node == nullptr) {
        return;
    }
    inorder(node->left);
    cout << node->key << " ";
    inorder(node->right);
}

void destroy(Node* node) {
    if (node == nullptr) {
        return;
    }
    destroy(node->left);
    destroy(node->right);
    delete node;
}

int main() {
    Node* root = nullptr;
    for (int k : {5, 3, 8, 1, 4}) {
        root = insert(root, k);
```

```cpp
struct Node {
    int key;
    Node* left;
    Node* right;

Node* insert(Node* root, int key) {
    if (root == nullptr) {
        return new Node{key, nullptr, nullptr};
    }
    if (key < root->key) {
        root->left = insert(root->left, key);
    } else {
        root->right = insert(root->right, key);
    }
    return root;
}

bool contains(const Node* node, int key) {
    if (node == nullptr) {
        return false;
    }
    if (key == node->key) {
        return true;
    }
    return key < node->key ? contains(node->left, key) : contains(node->right, key);
}

int height(const Node* node) {
    if (node == nullptr) {
        return 0;
    }

}
```


Compiler output (exactly as g++ prints it):

```text
semicolon.cpp:7:2: error: expected ';' after struct definition

      |  ^
      |  ;
```

Fixed code (the struct ends with a semicolon):

```cpp
#include <iostream>
using namespace std;

struct Point {
    int x;
    int y;
};

int main() {
    Point p{1, 2};
    cout << p.x << endl;
    return 0;
}
```

```text
1
```

**Why:** A struct or a class definition ends with a semicolon. The compiler often reports the error on the next line, so read one line up.
